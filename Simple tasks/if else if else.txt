import java.util.*

fun main(args: Array<String>) {
    val scan = Scanner(System.`in`)
    val temperature: Int = scan.nextLine().trim().toInt()

    if (temperature > 85) {
        println("High temperature")
    } else if (temperature < 50) {
        println("Low temperature")
    } else {
        println("Normal temperature")
    }
}
