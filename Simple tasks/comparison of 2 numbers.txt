import java.util.Scanner

fun main(args: Array<String>) {
    val reader = Scanner(System.`in`)
    
   	val num1 = reader.nextInt()
    val num2 = reader.nextInt()
    
	if (num1 > num2){
        print("greater")
    } else if (num1 == num2) {
        print("equal")
    } else {
    	print("less")
	}
}
