Программа читает целое число.
Если число от 1 до 9999 включительно → вывести сумму его цифр.
Если число вне диапазона или не является положительным целым → вывести "Invalid Input".

Пример:
Вход: 1234 → Выход: 10
Вход: 9999 → Выход: 36
Вход: -5 → Выход: "Invalid Input"

import java.util.*
fun main(args: Array<String>) {

    val scan = Scanner(System.`in`)
    val userInput = scan.nextLine().trim()

    try {
        val number = userInput.toInt()

        if (number in 1..9999) {
            var num = number
            var sum = 0
            while (num > 0) {
                sum += num % 10
                num /= 10
            }
            println(sum)
        } else {
            println("Invalid Input")
        }
    } catch (e: NumberFormatException) {
        println("Invalid Input")
    }
}
