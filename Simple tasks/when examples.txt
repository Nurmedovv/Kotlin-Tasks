fun main(){
    val (var1, op, var2) = readln().split(" ")

    val a = var1.toInt()
    val b = var2.toInt()

    when (op) {
        "+" -> println(a + b)
        "-" -> println(a - b)
        "*" -> println(a * b)
        else -> println("Unknown operator")
    }
}

fun main () {
    val fib = listOf("0", "1", "2", "3", "5", "8", "13", "21", "34", "55")
    val tri = listOf("0", "1", "3", "6", "10", "15", "21", "28", "36", "45")
    val p = listOf("1", "10", "100", "1000", "10000", "100000")
    println(when (readln()){
        in fib -> "F"
        in tri -> "T"
        in p -> "P"
        else -> "N"
    })
}

fun main() {
    val n = readln().toInt()
    println(
        when (n) {
        in 1..4 -> "few"
        in 5..9 -> "several"
        in 10..19 -> "pack"
        in 20..49 -> "lots"
        in 50..99 -> "horde"
        in 100..249 -> "throng"
        in 250..499 -> "swarm"
        in 500..999 -> "zounds"
        else -> if (n >= 1000) "legion" else "no army"
        }
    )
}

Программа читает целое число (от 1 до 4 цифр).
Нужно вывести количество цифр в этом числе.
Использовать when для проверки диапазонов.

Пример:
655 → 3
12 → 2

fun main() {
    val a = readln().toInt()
    println(
        when (a) {
            in 1..9 -> "1"
            in 10..99 -> "2"
            in 100..999 -> "3"
            in 1000..9999 -> "4"
            else -> "Error"
        }
    )
}
