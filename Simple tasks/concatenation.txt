Write a function concatenate that takes a list of words and a separator.
Return a single string with all words joined by the separator.
Use StringBuilder or buildString.

Example:
Words: ["one", "two", "three"], separator: " - "
Result: "one - two - three"

fun concatenate(list: List<String>, separator: String): String {
    return buildString {
        for ((index, element) in list.withIndex()) {
            append(element)
            if (index < list.lastIndex) {
                append(separator)
            }
        }
    }
}
